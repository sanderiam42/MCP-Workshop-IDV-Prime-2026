## start on your cloudshell prompt - setting up for rest of lab

# make sure your AWS session is sound

# this gets you to your actual lab machine for the first steps
aws ssm start-session --target <your_instance_id> --document-name AWS-StartInteractiveCommand --parameters command="cd ~ && exec bash -l"

## you are now on the main host machine for the lab where the docker containers will run
ls -al
cd MCP[TAB]

# this will apply your individual config to the files

# this now has the real commands to copy and paste - you can use this command multiple times as needed
cat ~/MCP-Workshop-IDV-Prime-2026/cheatsheet/copyPasta.txt

# this puts you into the main lab machine to use for actual MCP steps
docker exec -it __CLIENT_CONTAINER__ bash

# apply lab config inside the container

# make the XAA binary executable
