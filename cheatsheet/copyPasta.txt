### Bonus: step-by-step curl walk of the XAA flow (from inside the client container) - if the LLM is being testy but you want to play with the XAA flow anyway this is for you

## this performs the same three-step token chain the Lab 3 agent does, but by hand

# it reuses the Deadpool client you provisioned at the top of Lab 3 — paste the same client_id and client_secret values you put into mcp-config.json
export CLIENT_ID="<your-client-id-from-lab-3>"
export CLIENT_SECRET="<your-client-secret-from-lab-3>"

# now you create the BASIC AuthN to be used by the curl calls using those creds
export BASIC=$(echo -n "${CLIENT_ID}:${CLIENT_SECRET}" | base64)

# Step 1 — client credentials grant → ID token (machine identity, sub=client_id)
export ID_TOKEN=$(curl -s -X POST http://auth-server:8081/token \
  -H "Authorization: Basic ${BASIC}" \
  -H "Content-Type: application/x-www-form-urlencoded" \
  -d "grant_type=client_credentials" \
  -d "scope=mcp:read mcp:write" | jq -r '.id_token')

# Step 2 — token exchange → ID-JAG (cross-app authorization grant)
# audience/resource must match RESOURCE_SERVER_ORIGIN in .env (container mode)
export ID_JAG=$(curl -s -X POST http://auth-server:8081/token \
  -H "Authorization: Basic ${BASIC}" \
  -H "Content-Type: application/x-www-form-urlencoded" \
  -d "grant_type=urn:ietf:params:oauth:grant-type:token-exchange" \
  -d "subject_token=${ID_TOKEN}" \
  -d "subject_token_type=urn:ietf:params:oauth:token-type:id_token" \
  -d "requested_token_type=urn:ietf:params:oauth:token-type:id-jag" \
  -d "audience=http://resource-server:8082" \
  -d "resource=http://resource-server:8082/mcp" \
  -d "scope=mcp:read mcp:write" | jq -r '.access_token')

# Step 3 — JWT bearer → resource access token
export ACCESS_TOKEN=$(curl -s -X POST http://resource-server:8082/oauth/token \
  -H "Authorization: Basic ${BASIC}" \
  -H "Content-Type: application/x-www-form-urlencoded" \
  -d "grant_type=urn:ietf:params:oauth:grant-type:jwt-bearer" \
  -d "assertion=${ID_JAG}" \
  -d "scope=mcp:read mcp:write" | jq -r '.access_token')

# Step 4 — call an MCP tool directly with the access token
curl -s -X POST http://resource-server:8082/mcp \
  -H "Authorization: Bearer ${ACCESS_TOKEN}" \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"tools/call","params":{"name":"list_todos","arguments":{}}}' | jq .

# Same call via the requesting-app bridge — performs all three steps automatically
curl -s -X POST http://requesting-app:3000/host/mcp \
  -H "Content-Type: application/json" \
  -H "X-Demo-Client: ${CLIENT_ID}" \
  -H "X-Demo-Client-Secret: ${CLIENT_SECRET}" \
  -d '{"jsonrpc":"2.0","id":1,"method":"tools/call","params":{"name":"list_todos","arguments":{}}}' | jq .
